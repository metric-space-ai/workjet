import crypto from "node:crypto";

export class DeviceKeyError extends Error {
  constructor(public readonly code: string, public readonly status = 401) {
    super(code);
  }
}

export type DevicePublicJwk = { crv: "P-256"; kty: "EC"; x: string; y: string };

function canonicalBase64Url(value: unknown, bytes: number): value is string {
  return typeof value === "string" && /^[A-Za-z0-9_-]+$/.test(value)
    && Buffer.from(value, "base64url").length === bytes
    && Buffer.from(value, "base64url").toString("base64url") === value;
}

export function normalizeDevicePublicJwk(input: unknown): DevicePublicJwk {
  if (!input || typeof input !== "object" || Array.isArray(input)) {
    throw new DeviceKeyError("DEVICE_KEY_INVALID", 400);
  }
  const jwk = input as Record<string, unknown>;
  if (jwk.kty !== "EC" || jwk.crv !== "P-256" || "d" in jwk
    || (jwk.alg !== undefined && jwk.alg !== "ES256")
    || (jwk.use !== undefined && jwk.use !== "sig")
    || (jwk.key_ops !== undefined && (!Array.isArray(jwk.key_ops)
      || jwk.key_ops.length !== 1 || jwk.key_ops[0] !== "verify"))
    || !canonicalBase64Url(jwk.x, 32) || !canonicalBase64Url(jwk.y, 32)) {
    throw new DeviceKeyError("DEVICE_KEY_INVALID", 400);
  }
  // RFC 7638: required public members only, in lexicographic order.
  const publicJwk: DevicePublicJwk = { crv: "P-256", kty: "EC", x: jwk.x, y: jwk.y };
  try {
    const key = crypto.createPublicKey({ key: publicJwk, format: "jwk" });
    if (key.asymmetricKeyType !== "ec" || key.asymmetricKeyDetails?.namedCurve !== "prime256v1") {
      throw new Error("wrong curve");
    }
  } catch {
    throw new DeviceKeyError("DEVICE_KEY_INVALID", 400);
  }
  return publicJwk;
}

export function deviceKeyThumbprint(jwk: DevicePublicJwk) {
  return crypto.createHash("sha256").update(JSON.stringify(normalizeDevicePublicJwk(jwk))).digest("base64url");
}

export function deviceProofPayload(nonce: string, method: string, host: string, path: string, body: Uint8Array) {
  const bodyHash = crypto.createHash("sha256").update(body).digest("hex");
  return `${nonce}|${method.toUpperCase()}|${host}|${path}|${bodyHash}`;
}

export function verifyDeviceProof(jwk: DevicePublicJwk, signature: string, payload: string) {
  if (!canonicalBase64Url(signature, 64)) return false;
  try {
    return crypto.verify("sha256", Buffer.from(payload, "utf8"), {
      key: crypto.createPublicKey({ key: jwk, format: "jwk" }),
      dsaEncoding: "ieee-p1363",
    }, Buffer.from(signature, "base64url"));
  } catch {
    return false;
  }
}

export function validDeviceThumbprint(value: unknown): value is string {
  return canonicalBase64Url(value, 32);
}

export function validDeviceNonce(value: unknown): value is string {
  return canonicalBase64Url(value, 32);
}

export function hasDeviceProof(request: Request) {
  return ["x-ctox-device-thumbprint", "x-ctox-device-nonce", "x-ctox-device-signature"]
    .some((header) => request.headers.has(header));
}

// Hash the original bytes. Never stringify parsed JSON for signature checks.
export async function readDeviceRequestBody(request: Request, maximumBytes = 8192) {
  const reader = request.body?.getReader();
  if (!reader) return Buffer.alloc(0);
  const chunks: Uint8Array[] = [];
  let size = 0;
  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      size += value.length;
      if (size > maximumBytes) {
        await reader.cancel();
        throw new DeviceKeyError("DEVICE_REQUEST_TOO_LARGE", 413);
      }
      chunks.push(value);
    }
    return Buffer.concat(chunks);
  } finally {
    reader.releaseLock();
  }
}

export function parseDeviceJson(body: Uint8Array): Record<string, unknown> {
  try {
    const parsed = JSON.parse(Buffer.from(body).toString("utf8"));
    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) return parsed;
  } catch { /* Return one public input error. */ }
  throw new DeviceKeyError("DEVICE_REQUEST_INVALID", 400);
}
